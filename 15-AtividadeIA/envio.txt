CREATE TABLE configuracoes (
    id SERIAL PRIMARY KEY,
    chave VARCHAR(50) UNIQUE NOT NULL,
    valor_percentual DECIMAL(5,2) NOT NULL
);

CREATE TABLE regras_parcelamento (
    qtd_parcelas INT PRIMARY KEY,
    percentual_acrescimo DECIMAL(5,2) NOT NULL
);

CREATE TABLE clientes (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) UNIQUE,
    rg VARCHAR(20),
    endereco_completo VARCHAR(255),
    telefone VARCHAR(20)
);

CREATE TABLE vendedores (
    codigo SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cpf VARCHAR(14) UNIQUE NOT NULL,
    endereco_completo VARCHAR(255),
    telefone VARCHAR(20),
    celular VARCHAR(20),
    salario_base DECIMAL(10,2) NOT NULL
);

CREATE TABLE fornecedores (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    cnpj VARCHAR(20) UNIQUE NOT NULL,
    telefone VARCHAR(20)
);

CREATE TABLE produtos (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    descricao TEXT,
    preco_venda DECIMAL(10,2) NOT NULL,
    percentual_comissao DECIMAL(5,2) NOT NULL,
    -- Campos de controle (adicionados para resolver a Consulta 8)
    estoque_inicial INT NOT NULL DEFAULT 0,
    total_comprado INT NOT NULL DEFAULT 0,
    estoque_atual INT NOT NULL DEFAULT 0,
    estoque_minimo INT NOT NULL DEFAULT 5,
    qtd_ideal_recompra INT NOT NULL DEFAULT 10
);

CREATE TABLE fornecedor_produto (
    fornecedor_id INT REFERENCES fornecedores(id),
    produto_id INT REFERENCES produtos(id),
    tempo_entrega_dias INT NOT NULL,
    PRIMARY KEY (fornecedor_id, produto_id)
);

CREATE TABLE vendas (
    id SERIAL PRIMARY KEY,
    vendedor_id INT NOT NULL REFERENCES vendedores(codigo),
    cliente_id INT REFERENCES clientes(id),
    data_hora TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    total_venda DECIMAL(10,2) DEFAULT 0.00,
    tipo_pagamento VARCHAR(20) CHECK (tipo_pagamento IN ('A_VISTA', 'A_PRAZO')),
    status VARCHAR(20) DEFAULT 'EM_ABERTO'
);

CREATE TABLE itens_venda (
    id SERIAL PRIMARY KEY,
    venda_id INT NOT NULL REFERENCES vendas(id),
    produto_id INT NOT NULL REFERENCES produtos(id),
    quantidade INT NOT NULL,
    preco_unitario DECIMAL(10,2) NOT NULL,
    valor_comissao DECIMAL(10,2) DEFAULT 0.00
);

CREATE TABLE parcelas_venda (
    id SERIAL PRIMARY KEY,
    venda_id INT NOT NULL REFERENCES vendas(id),
    numero_parcela INT NOT NULL,
    data_vencimento DATE NOT NULL,
    valor_parcela DECIMAL(10,2) NOT NULL,
    data_pagamento DATE,
    valor_pago DECIMAL(10,2)
);

    ((v.id % 20) + 1), -- Distribui itens entre os 20 produtos

    p.preco_venda,

-- 3. Maiores Vendedores do Mês
SELECT 
    v.nome AS vendedor, 
    SUM(ven.total_venda) AS total_vendido
FROM vendedores v
JOIN vendas ven ON v.codigo = ven.vendedor_id
WHERE EXTRACT(MONTH FROM ven.data_hora) = 10 
  AND EXTRACT(YEAR FROM ven.data_hora) = 2023
GROUP BY v.nome
ORDER BY total_vendido DESC
LIMIT 3;

-- 4. Clientes que Mais Gastaram na História
SELECT 
    c.nome AS cliente, 
    SUM(ven.total_venda) AS total_gasto
FROM clientes c
JOIN vendas ven ON c.id = ven.cliente_id
GROUP BY c.id, c.nome
ORDER BY total_gasto DESC
LIMIT 3;

-- 5. Salário + Comissão por Vendedor
SELECT 
    v.nome AS vendedor, 
    v.salario_base,
    COALESCE(SUM(iv.valor_comissao), 0) AS total_comissoes,
    (v.salario_base + COALESCE(SUM(iv.valor_comissao), 0)) AS salario_final
FROM vendedores v
LEFT JOIN vendas ven ON v.codigo = ven.vendedor_id
LEFT JOIN itens_venda iv ON ven.id = iv.venda_id
GROUP BY v.codigo, v.nome, v.salario_base
ORDER BY salario_final DESC;

-- 6. Relatório de Reposição de Estoque
SELECT 
    id AS codigo_produto, 
    nome AS produto, 
    estoque_atual, 
    estoque_minimo, 
    qtd_ideal_recompra AS quantidade_comprar
FROM produtos
WHERE estoque_atual <= estoque_minimo
ORDER BY estoque_atual ASC;

-- 7. Resumo de Vendas em Dinheiro (À Vista) x A Prazo
SELECT 
    tipo_pagamento, 
    SUM(total_venda) AS faturamento_total
FROM vendas
WHERE data_hora >= '2023-10-01 00:00:00' 
  AND data_hora <= '2023-10-31 23:59:59'
GROUP BY tipo_pagamento;

--8. Auditoria de Estoque (Identificação de Perdas/Furtos)
SELECT 
    p.id AS codigo,
    p.nome AS produto,
    p.estoque_inicial,
    p.total_comprado AS compras_realizadas,
    COALESCE(SUM(iv.quantidade), 0) AS unidades_vendidas,
    p.estoque_atual AS saldo_registrado,
    (p.estoque_inicial + p.total_comprado - COALESCE(SUM(iv.quantidade), 0)) AS saldo_matematico_correto,
    CASE 
        WHEN p.estoque_atual = (p.estoque_inicial + p.total_comprado - COALESCE(SUM(iv.quantidade), 0)) 
            THEN 'ESTOQUE OK'
        WHEN p.estoque_atual < (p.estoque_inicial + p.total_comprado - COALESCE(SUM(iv.quantidade), 0)) 
            THEN 'ALERTA: PRODUTO SUMIU'
        ELSE 'ALERTA: SOBRA INEXPLICADA'
    END AS status_auditoria
FROM produtos p
LEFT JOIN itens_venda iv ON p.id = iv.produto_id
GROUP BY 
    p.id, 
    p.nome, 
    p.estoque_inicial, 
    p.total_comprado, 
    p.estoque_atual
ORDER BY p.id;
